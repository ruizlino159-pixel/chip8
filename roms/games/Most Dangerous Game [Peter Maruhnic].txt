    <script>
        // --- LA CPU VIRTUAL ---
        class Chip8 {
            constructor() {
                this.reset();
            }

            reset() {
                // 4KB de Memoria RAM
                this.memory = new Uint8Array(4096);
                // 16 Registros de 8-bits (V0 a VF)
                this.v = new Uint8Array(16);
                // Registro de índice I
                this.i = 0;
                // Program Counter (PC) - Empieza en 0x200 (estándar Chip-8)
                this.pc = 0x200;
                
                // Pantalla (64x32 píxeles, blanco y negro)
                this.display = new Array(64 * 32).fill(0);
                
                // Pila (Stack) para subrutinas
                this.stack = [];
                
                // Temporizadores
                this.delayTimer = 0;
                this.soundTimer = 0;
                
                this.paused = true;
            }

            loadRom(romData) {
                this.reset();
                // Cargar la ROM en la memoria empezando en 0x200
                for (let i = 0; i < romData.length; i++) {
                    this.memory[0x200 + i] = romData[i];
                }
                this.paused = false;
                console.log("ROM cargada. Iniciando CPU...");
            }

            // EL CICLO DE LA CPU (FETCH - DECODE - EXECUTE)
            cycle() {
                if (this.paused) return;

                // 1. FETCH (Buscar): Leer 2 bytes de la memoria para formar el Opcode
                // Ejemplo: 0xA2F0 (Un opcode son 16 bits)
                let opcode = (this.memory[this.pc] << 8) | this.memory[this.pc + 1];

                // 2. DECODE & EXECUTE (Decodificar y Ejecutar)
                this.executeOpcode(opcode);

                // Avanzar el Program Counter (PC)
                this.pc += 2;

                // Decrementar timers
                if (this.delayTimer > 0) this.delayTimer--;
                if (this.soundTimer > 0) this.soundTimer--;
            }

            executeOpcode(opcode) {
                // Aquí es donde ocurre la magia. Hay que implementar 35 instrucciones.
                // Usamos máscaras de bits para entender qué pide el juego.
                
                // Ejemplo: Limpiar pantalla (00E0)
                if (opcode === 0x00E0) {
                    this.display.fill(0);
                    return;
                }

                // Ejemplo: Salto (Jump) - 1NNN
                // Si el primer nibble (4 bits) es 1, saltamos a la dirección NNN
                if ((opcode & 0xF000) === 0x1000) {
                    this.pc = (opcode & 0x0FFF);
                    this.pc -= 2; // Compensar el avance del ciclo
                    return;
                }

                // Ejemplo: Set Register - 6XNN (V[x] = NN)
                if ((opcode & 0xF000) === 0x6000) {
                    let x = (opcode & 0x0F00) >> 8;
                    let value = (opcode & 0x00FF);
                    this.v[x] = value;
                    return;
                }
                
                // Ejemplo: Add Value - 7XNN (V[x] += NN)
                if ((opcode & 0xF000) === 0x7000) {
                    let x = (opcode & 0x0F00) >> 8;
                    let value = (opcode & 0x00FF);
                    this.v[x] += value;
                    return;
                }

                // Aquí faltarían el resto de instrucciones...
                console.log(`Opcode no implementado: ${opcode.toString(16)}`);
            }
        }

        // --- SISTEMA PRINCIPAL (Main Loop) ---
        const chip8 = new Chip8();
        const canvas = document.getElementById('display');
        const ctx = canvas.getContext('2d');
        const scale = 10; // Escalar píxeles para verlos mejor

        function draw() {
            // Limpiar canvas
            ctx.fillStyle = '#000';
            ctx.fillRect(0, 0, canvas.width, canvas.height);

            // Dibujar píxeles activos
            ctx.fillStyle = '#0f0'; // Color verde hacker
            for (let i = 0; i < chip8.display.length; i++) {
                if (chip8.display[i] === 1) {
                    let x = (i % 64) * scale;
                    let y = Math.floor(i / 64) * scale;
                    ctx.fillRect(x, y, scale, scale);
                }
            }
        }

        function step() {
            // Ejecutar varios ciclos por frame para velocidad (aprox 60hz * 10)
            for(let i=0; i<10; i++) {
                chip8.cycle();
            }
            draw();
            requestAnimationFrame(step);
        }

        // --- CARGADOR DE ARCHIVOS ---
        document.getElementById('romInput').addEventListener('change', function(e) {
            const file = e.target.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                const buffer = e.target.result;
                const romData = new Uint8Array(buffer);
                chip8.loadRom(romData);
                step(); // Iniciar bucle
            };
            reader.readAsArrayBuffer(file);
        });
